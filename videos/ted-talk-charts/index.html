def render_frame(t, W=1920, H=1080, fi=0):
    img = paper(W, H)
    cv = Cv(img, W, H)
    tp = T_PILL
    if not cv.vert and not C.STATE["clean"]:   # the masthead steps aside while the subscribe pill is up (16:9)
        cv.op = 1 - sstep(tp - 0.1, tp + 0.3, t) * (1 - sstep(tp + 4.6, tp + 5.0, t))
    if not (T_PLACE + 0.6 < t):
        masthead(cv, t)
    else:
        cv.dy = -drift_cam(cv, t); masthead(cv, t); cv.dy = 0.0
    cv.op = 1.0
    draw_figs(cv, t)
    if T_SOW - 0.3 < t < T_DIM + 0.6:      # the zoomed-out page slides under a fixed masthead
        yb = int(round((186 if not cv.vert else 372) * cv.k)); fe = max(2, int(14 * cv.k))
        a = np.clip((yb - np.arange(yb + fe, dtype=F32)) / fe + 1.0, 0, 1)[:, None, None]
        img[:yb + fe] = img[:yb + fe] * (1 - a) + paper(W, H)[:yb + fe] * a
        masthead(cv, t)
    fig_so(cv, t)
    light(img, t, W, H)
    lyrics(img, t, W, H)
    tabloid(img, t, W, H)
    if not C.STATE["clean"]:
        C.subscribe_pill(img, t, W, H, T_PILL, 5.0)
        C.end_screen(img, t, W, H, T_END, DUR - T_END, "It's always a Ted talk.", sub_line="Subscribe for a new song every week",
                     title="TED Talk, in charts", accent=RED)
    return finish(img, t, fi)

What you see

The whole song is one issue of a newspaper called The Daily Standup, with a new chart every few seconds: how Ted handles a question (it never reaches ANSWER), my will to live by minute (it falls off a cliff at minute four), the topics he covered (your question: 0%), his most-used words (so: 214, the file: 0), buzzword bingo, and a zoom-out on the one question nobody answered. When somebody dims the lights the paper sits under a spotlight; at Dana's hard stop the office lights snap back on and the paper prints a CORRECTION. In the end Ted's dot unplugs from the team network and drifts down into a footnote.
